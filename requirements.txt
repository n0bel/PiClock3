# Python 3.9 (Raspberry Pi OS Bullseye) is the floor.

# tzlocal and metar are deliberately NOT capped.  Their newer releases declare
# requires_python >= 3.10, so pip already resolves tzlocal 5.3.1 / metar 1.11
# on Bullseye and tzlocal 5.4.4 / metar 2.0.1 on Bookworm and newer, by itself.
# Capping them would hold new systems on old packages for no reason.
tzlocal>=3.0
metar>=1.11
astral~=3.2
compassheadinglib~=2026.1.22

# pyyaml-include IS capped, for a different reason.  2.x renamed the module
# from yamlinclude to yaml_include and changed the api, and it installs quite
# happily on 3.9 - requires_python cannot express an api break, so the cap has
# to.  Config.py uses the 1.x YamlIncludeConstructor.
pyyaml-include>=1.3,<2

# PyQt5 is deliberately not listed.  On a Pi install it with
#   sudo apt install python3-pyqt5
# pip would build Qt from source, which takes hours and usually runs out
# of memory on a Pi.
